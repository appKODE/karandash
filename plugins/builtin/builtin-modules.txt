# Built-in module registry manifest
#
# Single source of truth for all modules bundled with the Karandash CLI.
# The Gradle task `generateBuiltInRegistry` reads this file and produces
# GeneratedBuiltInRegistry.kt with the factory list.
#
# Format: one fully-qualified factory expression per line.
#   - Simple:  com.example.pkg.MyModule           -> { _ -> MyModule() }
#   - Wrapped: com.example.Adapter(com.example.Inner())  -> { _ -> Adapter(Inner()) }
#   - Registry: com.example.pkg.MyModule @registry -> { registry -> MyModule(registry) }
#
# Append @registry to modules that need the ModuleRegistry as a constructor parameter.
#
# Blank lines and lines starting with # are ignored.
# ORDER MATTERS — modules are loaded in the order listed here.
